#!/usr/bin/env python3
"""Render the original fictional count chart; only the standard library is used."""
import argparse
from fractions import Fraction
from html import escape
import json
from pathlib import Path

STEM = "starting-count-and-control-denominators-v1"
PALETTES = {
    "light": {
        "background": "#faf7f1", "ink": "#263640", "muted": "#465967",
        "grid": "#d8ddd9", "control": "#4c7796", "treated": "#bb637a",
        "hatch": "#733a4b", "card": "#f0ece5", "inside": "#ffffff",
    },
    "dark": {
        "background": "#172530", "ink": "#f1eee8", "muted": "#c8d3da",
        "grid": "#40515c", "control": "#7fa9c5", "treated": "#d18ca0",
        "hatch": "#713e50", "card": "#213441", "inside": "#13222d",
    },
}


def number(value):
    """Write exact terminating coordinates without binary float drift."""
    value = Fraction(value)
    if value.denominator == 1:
        return str(value.numerator)
    scaled = value * 1000
    assert scaled.denominator == 1, "Only exact three-place geometry is needed."
    return f"{scaled.numerator / 1000:.3f}".rstrip("0").rstrip(".")


def render(theme, data, coordinates, copy):
    palette = PALETTES[theme]
    assert coordinates["nativeViewBox"] == [0, 0, 600, 1190]
    assert [(x["condition"], x["count"]) for x in data["observations"]] == [
        ("Control", 1000), ("Treated", 1000), ("Control", 2000), ("Treated", 1000)
    ]
    assert [x["width"] for x in coordinates["bars"]] == [252, 252, 504, 252]
    assert Fraction(1000, 2000) == Fraction(1, 2)
    assert Fraction(1000, 1000) == 1
    svg = [
        '<?xml version="1.0" encoding="UTF-8"?>',
        '<svg xmlns="http://www.w3.org/2000/svg" width="600" height="1190" '
        'viewBox="0 0 600 1190" role="img" aria-labelledby="chart-title chart-desc">',
        '<title id="chart-title">Half the later control, equal to the starting count</title>',
        f'<desc id="chart-desc">{escape(copy["alt"])}</desc>',
        f'<rect width="600" height="1190" fill="{palette["background"]}"/>',
        '<defs>',
        '<pattern id="treated-hatch" width="12" height="12" patternUnits="userSpaceOnUse">',
        f'<path d="M -3 3 L 3 -3 M 0 12 L 12 0 M 9 15 L 15 9" '
        f'fill="none" stroke="{palette["hatch"]}" stroke-width="2"/>',
        '</pattern>',
        '</defs>',
        f'<g font-family="Arial, Helvetica, sans-serif" fill="{palette["ink"]}">',
    ]

    def text(x, y, content, size=28, weight=400, anchor="start", fill=None, extra=""):
        svg.append(
            f'<text x="{number(x)}" y="{number(y)}" font-size="{size}" '
            f'font-weight="{weight}" text-anchor="{anchor}"'
            + (f' fill="{fill}"' if fill else "")
            + (f' {extra}' if extra else "")
            + f'>{escape(content)}</text>'
        )

    text(36, 66, "50% of control.", 36, 700)
    text(36, 110, "100% of start.", 36, 700)
    text(36, 154, "Fictional viable-cell counts", 26, fill=palette["muted"])
    text(48, 205, "Start", 32, 700)
    text(48, 514, "Later", 32, 700)

    for observation, bar, label_y, count_y in zip(
        data["observations"], coordinates["bars"],
        coordinates["conditionLabelYs"], coordinates["countLabelYs"],
    ):
        condition = observation["condition"]
        count = observation["count"]
        expected_width = Fraction(count) * Fraction(504, 2000)
        assert expected_width == bar["width"]
        x, y, width, height = bar["x"], bar["y"], bar["width"], bar["height"]
        text(x, label_y, condition, 28, 600)
        svg.append(
            f'<rect id="bar-{observation["id"]}" data-count="{count}" '
            f'x="{number(x)}" y="{number(y)}" width="{number(width)}" '
            f'height="{number(height)}" fill="{palette[condition.lower()]}"/>'
        )
        if condition == "Treated":
            svg.append(
                f'<rect x="{number(x)}" y="{number(y)}" width="{number(width)}" '
                f'height="{number(height)}" fill="url(#treated-hatch)"/>'
            )
        if count == 2000:
            text(538, count_y, "2,000", 28, 700, "end", palette["inside"])
        else:
            text(x + width + 12, count_y, "1,000", 28, 700)

    for axis_y in coordinates["axisYs"]:
        svg.append(
            f'<path d="M 48 {axis_y} H 552" fill="none" '
            f'stroke="{palette["muted"]}" stroke-width="2"/>'
        )
        for tick in coordinates["ticks"]:
            x = tick["x"]
            svg.append(
                f'<path d="M {x} {axis_y} V {axis_y + 9}" fill="none" '
                f'stroke="{palette["muted"]}" stroke-width="2"/>'
            )
            text(x, axis_y + 36, f'{tick["count"]:,}', 26, anchor="middle")
        text(300, axis_y + 64, "Viable cells • same scale", 26, anchor="middle", fill=palette["muted"])

    for comparison, card in zip(data["comparisons"], coordinates["denominatorCards"]):
        x, y, width, height = card["x"], card["y"], card["width"], card["height"]
        svg.append(
            f'<rect x="{x}" y="{y}" width="{width}" height="{height}" '
            f'rx="12" fill="{palette["card"]}"/>'
        )
        cx = x + Fraction(width, 2)
        text(cx, y + 42, "Compare with", 26, anchor="middle")
        label = "later control" if comparison["id"] == "relative-to-later-control" else "treated start"
        text(cx, y + 77, label, 28, 700, "middle")
        formula = f'{comparison["numeratorCount"]:,} ÷ {comparison["denominatorCount"]:,}'
        text(cx, y + 125, formula, 26, anchor="middle")
        text(cx, y + 190, f'= {comparison["percentage"]}%', 44, 700, "middle")
    text(300, 1150, "Net counts are not cell fates.", 27, 600, "middle")
    svg.extend(["</g>", "</svg>"])
    return "\n".join(svg) + "\n"


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--inputs", type=Path, default=Path(__file__).parent)
    parser.add_argument("--output-dir", type=Path, required=True)
    args = parser.parse_args()
    data = json.loads((args.inputs / "chart-data.json").read_text())
    coordinates = json.loads((args.inputs / "chart-coordinates.json").read_text())
    copy = json.loads((args.inputs / "proposed-copy.json").read_text())
    args.output_dir.mkdir(parents=True, exist_ok=True)
    for theme in PALETTES:
        (args.output_dir / f"{STEM}-{theme}.svg").write_text(render(theme, data, coordinates, copy))


if __name__ == "__main__":
    main()
END_LITERAL_RENDERER_PY
