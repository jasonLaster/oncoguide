```python
#!/usr/bin/env python3
"""Original editable denominator diagram. Python standard library only."""
from pathlib import Path
from fractions import Fraction
import argparse, hashlib, html, json

STEM = 'tissue-input-and-recovered-denominators-v1'

def digest(b):
    return hashlib.sha256(b).hexdigest()

def render(theme, data, coordinates, style, alt):
    c = style[theme + 'Palette']
    out = ['<?xml version="1.0" encoding="UTF-8"?>',
           '<svg xmlns="http://www.w3.org/2000/svg" width="600" height="1860" viewBox="0 0 600 1860" role="img" aria-labelledby="title description">',
           '<title id="title">Follow the denominator</title>',
           '<desc id="description">' + html.escape(alt) + '</desc>']
    def rect(x, y, w, h, fill, stroke=None, rx=16, sw=2, dashed=False):
        line = '' if stroke is None else f' stroke="{c[stroke]}" stroke-width="{sw}"'
        dash = ' stroke-dasharray="8 6"' if dashed else ''
        out.append(f'<rect x="{x}" y="{y}" width="{w}" height="{h}" rx="{rx}" fill="{c[fill]}"{line}{dash}/>')
    def path(d, fill='none', stroke='line', sw=2):
        color = 'none' if fill == 'none' else c[fill]
        edge = '' if stroke is None else f' stroke="{c[stroke]}" stroke-width="{sw}"'
        out.append(f'<path d="{d}" fill="{color}"{edge} stroke-linecap="round" stroke-linejoin="round"/>')
    def text(x, y, lines, size=28, bold=False, anchor='start'):
        out.append(f'<text x="{x}" y="{y}" fill="{c["ink"]}" font-family="Arial, Helvetica, sans-serif" font-size="{size}" font-weight="{700 if bold else 400}" text-anchor="{anchor}">')
        for i, line in enumerate(lines):
            out.append(f'<tspan x="{x}" dy="{0 if i == 0 else size + 7}">{html.escape(line)}</tspan>')
        out.append('</text>')
    def ellipse(cx, cy, rx, ry, fill='neutral', sw=3):
        out.append(f'<ellipse cx="{cx}" cy="{cy}" rx="{rx}" ry="{ry}" fill="{c[fill]}" stroke="{c["line"]}" stroke-width="{sw}"/>')
    def leader(start, end):
        path(f'M {start[0]} {start[1]} L {end[0]} {end[1]}', sw=2.5)
        out.append(f'<circle cx="{end[0]}" cy="{end[1]}" r="3.5" fill="{c["ink"]}"/>')
    def arrow(x1, y1, x2, y2):
        path(f'M {x1} {y1} L {x2} {y2-5}', sw=3)
        path(f'M {x2-7} {y2-12} L {x2} {y2} L {x2+7} {y2-12}', sw=3)

    rect(0, 0, 600, 1860, 'background', rx=0)
    text(300, 58, ['Follow the denominator'], 36, True, 'middle')
    text(300, 103, ['Three views, three denominators'], 28, False, 'middle')
    for panel in coordinates['panels']:
        rect(*panel['rect'], 'panel', 'line', rx=23)
        title = {'tissue-area':'1  Tissue area', 'input-nuclei':'2  Input nuclei', 'recovered-profiles':'3  Recovered profiles'}[panel['id']]
        text(*panel['title'], [title], 34, True)
    p = coordinates['panels'][0]
    # The border is the assessed bed area; rose patches are occupied areas, not cells.
    rect(*p['areaOutline'], 'matrix', 'line', rx=18, sw=3)
    path('M 110 247 C 142 230 161 254 181 249 C 205 244 230 264 226 284 C 225 306 192 312 179 302 C 150 313 139 298 121 294 C 98 288 95 261 110 247 Z', 'rose', None)
    path('M 83 338 C 98 329 121 341 116 356 C 112 367 88 366 83 354 Z', 'rose', None)
    for label in p['labels']:
        text(*label['position'], label['lines'], bold=label['position'][1] == 417)
    for l in p['leaders']:
        leader(l['start'], l['end'])
    # Deliberately no arrow from tissue-area cellularity to the input-nucleus fraction.
    text(80, 512, ['≠'], 44, True, 'middle')
    text(343, 508, ['Different denominator'], 28, True, 'middle')
    p = coordinates['panels'][1]
    for cx, cy in p['nucleusCenters']:
        ellipse(cx, cy, p['nucleusRadius'], 22)
        ellipse(cx-4, cy+3, 5, 6, 'panel', sw=1.5)
    for label in p['labels']:
        text(*label['position'], label['lines'], 30 if label['position'][0] == 279 else 28, label['position'][0] == 279)
    # All nucleus icons use the same neutral color; they cannot encode a malignant fraction.
    rect(*coordinates['uncertainProcess']['rect'], 'uncertain', 'line', rx=17, dashed=True)
    text(300, 945, ['Recovery + filtering'], 30, True, 'middle')
    text(300, 982, ['May change representation'], 28, False, 'middle')
    for ends in coordinates['uncertainProcess']['arrows']:
        arrow(*ends)
    p = coordinates['panels'][2]
    text(*p['definitionLabel']['position'], p['definitionLabel']['lines'])
    for x, y, w, h in p['profileCards']:
        rect(x, y, w, h, 'background', 'line', rx=10)
        text(x+w/2, y+36, ['Profile'], 28, False, 'middle')
        # Identical decorative rules, with no gene identities, bar heights or measurements.
        for yy in [y+56, y+70]:
            path(f'M {x+19} {yy} L {x+w-19} {yy}', stroke='neutral', sw=4)
    for label in p['labels']:
        text(*label['position'], label['lines'], bold=True)
    rect(*p['arithmeticInset'], 'uncertain', 'line', rx=17)
    for i, label in enumerate(p['arithmeticLabels']):
        text(*label['position'], label['lines'], size=34 if i == 3 else 28, bold=i in [0,3])
    text(*coordinates['illustrativeSymbolsLabel']['position'], [coordinates['illustrativeSymbolsLabel']['text']], 26, False, 'middle')
    out.append('</svg>')
    return ('\n'.join(out)+'\n').encode('utf-8')

def main():
    parser=argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--output-dir', required=True, type=Path)
    args=parser.parse_args()
    base=Path(__file__).resolve().parent
    data=json.loads((base/'data.json').read_bytes())
    coordinates=json.loads((base/'coordinate-plan.json').read_bytes())
    style=json.loads((base/'style.json').read_bytes())
    brief=json.loads((base/'brief.json').read_bytes())
    assert Fraction(data['inputMalignantFraction']['numerator'],data['inputMalignantFraction']['denominator'])*data['retainedProfilesAssumed']==data['expectedMalignantProfiles']==240
    assert coordinates['viewBox']==[0,0,600,1860]
    args.output_dir.mkdir(parents=True, exist_ok=True)
    names=[STEM+'-'+t+'.svg' for t in ['light','dark']]+['rendered-coordinate-plan.json','render-manifest.json']
    assert all(not (args.output_dir/n).exists() for n in names), 'Output paths must be fresh.'
    records=[]
    for theme in ['light','dark']:
        name=STEM+'-'+theme+'.svg';b=render(theme,data,coordinates,style,brief['alt'])
        (args.output_dir/name).write_bytes(b)
        records.append({'name':name,'sha256':digest(b),'bytes':len(b),'width':600,'height':1860})
    coordinate_bytes=(json.dumps(coordinates,indent=2,ensure_ascii=False)+'\n').encode()
    (args.output_dir/'rendered-coordinate-plan.json').write_bytes(coordinate_bytes)
    manifest={'renderer':'Python standard library original editable SVG; no raster or image model','inputs':{name:digest((base/name).read_bytes()) for name in ['render-diagram.py','data.json','coordinate-plan.json','style.json','brief.json','editable-diagram-specification-draft.txt']},'outputs':records+[{'name':'rendered-coordinate-plan.json','sha256':digest(coordinate_bytes),'bytes':len(coordinate_bytes)}]}
    (args.output_dir/'render-manifest.json').write_text(json.dumps(manifest,indent=2,ensure_ascii=False)+'\n')
    print(json.dumps(manifest,indent=2,ensure_ascii=False))
if __name__=='__main__':
    main()
```
