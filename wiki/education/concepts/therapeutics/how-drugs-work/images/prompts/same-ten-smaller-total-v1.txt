def number(value: int | float | Fraction) -> str:
    """Stable decimal coordinates; geometry derives from exact rational data."""
    if isinstance(value, int):
        return str(value)
    return f"{float(value):.9f}".rstrip("0").rstrip(".")


def validate(data: dict, coords: dict) -> dict:
    before, after, removed = data["before"], data["after"], data["removed"]
    assert (before["sensitive"], before["resistant"]) == (990, 10)
    assert (after["sensitive"], after["resistant"]) == (99, 10)
    assert before["total"] == before["sensitive"] + before["resistant"] == 1000
    assert after["total"] == after["sensitive"] + after["resistant"] == 109
    assert removed["sensitive"] == before["sensitive"] - after["sensitive"] == 891
    assert removed["resistant"] == 0
    assert removed["total"] == 891
    assert Fraction(after["sensitive"], before["sensitive"]) == Fraction(1, 10)
    fractions = [Fraction(r["resistant"], r["total"]) for r in (before, after)]
    assert fractions == [Fraction(1, 100), Fraction(10, 109)]
    assert data["fractions"] == ["10/1,000 = 1%", "10/109 ≈ 9.2%"]
    assert data["plotAxes"]["absoluteCount"]["min"] == 0
    assert data["plotAxes"]["absoluteCount"]["max"] == 1000
    assert data["plotAxes"]["absoluteCount"]["ticks"] == [0, 250, 500, 750, 1000]
    assert data["plotAxes"]["resistantFraction"]["min"] == 0
    assert data["plotAxes"]["resistantFraction"]["max"] == 10
    assert data["plotAxes"]["resistantFraction"]["ticks"] == [0, 5, 10]
    assert coords["canvas"] == {"width": WIDTH, "height": HEIGHT, "svgViewBox": "0 0 600 1440"}
    for k in ("countPlot", "fractionPlot"):
        assert (coords[k]["xMin"], coords[k]["xMax"]) == (96, 552)
    assert coords["countPlot"]["rowCenters"] == {"before": 346, "after": 510}
    assert coords["fractionPlot"]["rowCenters"] == {"before": 896, "after": 1046}
    count_length = Fraction(456)
    percent_length = Fraction(456)
    count_rows = []
    fraction_rows = []
    for name, state, frac in zip(("before", "after"), (before, after), fractions):
        sensitive_width = count_length * state["sensitive"] / 1000
        resistant_width = count_length * state["resistant"] / 1000
        resistant_start = Fraction(96) + sensitive_width
        count_rows.append({"state": name, "sensitive": state["sensitive"], "resistant": state["resistant"], "total": state["total"],
                           "sensitiveStartX": 96, "sensitiveWidth": float(sensitive_width),
                           "resistantStartX": float(resistant_start), "resistantWidth": float(resistant_width),
                           "resistantLeaderEndpointX": float(resistant_start + resistant_width / 2),
                           "totalEndX": float(resistant_start + resistant_width)})
        percent = frac * 100
        fraction_rows.append({"state": name, "numerator": state["resistant"], "denominator": state["total"],
                              "exactFraction": str(frac), "exactPercent": str(percent), "barWidth": float(percent_length * percent / 10)})
    assert count_rows[0]["resistantWidth"] == count_rows[1]["resistantWidth"] == 4.56
    assert abs(fraction_rows[1]["barWidth"] - coords["fractionPlot"]["afterBarWidth"]) < 1e-9
    return {"id": "next-gap-011", "fictional": True, "counts": count_rows, "fractions": fraction_rows,

class Drawing:
    def __init__(self, palette: dict):
        self.p = palette
        self.lines = [f'<svg xmlns="http://www.w3.org/2000/svg" width="{WIDTH}" height="{HEIGHT}" viewBox="0 0 {WIDTH} {HEIGHT}" role="img" aria-labelledby="chart-title chart-description">',
                      '<title id="chart-title">A larger fraction, not more resistant cells</title>',
                      '<desc id="chart-description">Fictional selection toy. Before: 990 sensitive plus 10 resistant cells, total 1,000. After: 99 sensitive plus the same 10 resistant cells, total 109. Both count bars use one linear 0 to 1,000 cell axis. A separate 0 to 10 percent axis shows 10/1,000 equals 1 percent and 10/109 approximately 9.2 percent. No division, state change, new mutation or clinical treatment effect is established.</desc>',
                      '<defs>',
                      f'<pattern id="resistant-hatch" width="10" height="10" patternUnits="userSpaceOnUse"><path d="M -2 2 L 2 -2 M 0 10 L 10 0 M 8 12 L 12 8" fill="none" stroke="{palette["hatch"]}" stroke-width="1.5"/></pattern>',
                      '</defs>',
                      f'<g font-family="Arial, DejaVu Sans, sans-serif" fill="{palette["ink"]}">']
    def rect(self, x, y, width, height, fill, stroke=None, radius=0, element_id=None):
        attrs = f' x="{number(x)}" y="{number(y)}" width="{number(width)}" height="{number(height)}" fill="{fill}"'
        if stroke:
            attrs += f' stroke="{stroke}" stroke-width="1.5"'
        if radius:
            attrs += f' rx="{number(radius)}"'
        if element_id:
            attrs += f' id="{escape(element_id, quote=True)}"'
        self.lines.append(f'<rect{attrs}/>')
    def text(self, x, y, value, size=28, anchor="start", weight="normal", quiet=False):
        color = self.p["quiet"] if quiet else self.p["ink"]
        self.lines.append(f'<text x="{number(x)}" y="{number(y)}" font-size="{size}" text-anchor="{anchor}" font-weight="{weight}" fill="{color}">{escape(value)}</text>')
    def path(self, points, stroke=None, width=1.5, element_id=None):
        d = "M " + " L ".join(f"{number(x)} {number(y)}" for x, y in points)
        identity = f' id="{escape(element_id, quote=True)}"' if element_id else ""
        self.lines.append(f'<path{identity} d="{d}" fill="none" stroke="{stroke or self.p["quiet"]}" stroke-width="{number(width)}"/>')
    def finish(self):
        return "\n".join(self.lines + ['</g>', '</svg>', '']).encode("utf-8")


def render(theme: str, data: dict, coords: dict, checks: dict) -> bytes:
    p = PALETTES[theme]
    d = Drawing(p)
    d.rect(0, 0, WIDTH, HEIGHT, p["ground"])
    d.text(40, 62, "A larger fraction,", 34, weight="bold")
    d.text(40, 106, "not more resistant cells", 34, weight="bold")
    d.text(40, 152, "Fictional selection toy", 28, quiet=True)
    d.rect(24, 181, 552, 514, p["panel"], p["border"], 16)
    d.text(48, coords["countPlot"]["headerY"], "Absolute cell counts", 30, weight="bold")
    d.text(48, 248, "Cells: same 0–1,000 axis", 28)
    count = coords["countPlot"]
    xmin, xmax = count["xMin"], count["xMax"]
    for tick in data["plotAxes"]["absoluteCount"]["ticks"]:
        x = Fraction(xmin) + Fraction(xmax - xmin) * tick / 1000
        d.path([(x, 310), (x, count["axisY"])], p["grid"], 1)
        d.path([(x, count["axisY"]), (x, count["axisY"] + 7)], p["quiet"], 1.5)
        d.text(x, count["axisY"] + 38, f"{tick:,}", 26, anchor="middle")
    d.path([(xmin, count["axisY"]), (xmax, count["axisY"])])
    for name, state, row in zip(("before", "after"), (data["before"], data["after"]), checks["counts"]):
        y = count["rowCenters"][name]
        h = count["barHeight"]
        heading_y = count["beforeHeadingY" if name == "before" else "afterHeadingY"]
        label_y = count["beforeExactLabelY" if name == "before" else "afterExactLabelY"]
        d.text(xmin, heading_y, name.capitalize(), 28, weight="bold")
        d.text(xmax, heading_y, f'Total {state["total"]:,}', 28, anchor="end")
        sensitive_width = Fraction(xmax - xmin) * state["sensitive"] / 1000
        resistant_width = Fraction(xmax - xmin) * state["resistant"] / 1000
        rx = Fraction(xmin) + sensitive_width
        d.rect(xmin, y - h / 2, sensitive_width, h, p["sensitive"], p["sensitiveEdge"], element_id=f"{name}-sensitive-count")
        d.rect(rx, y - h / 2, resistant_width, h, p["resistant"], element_id=f"{name}-resistant-count")
        d.rect(rx, y - h / 2, resistant_width, h, "url(#resistant-hatch)")
        d.text(xmin, label_y, f'{state["sensitive"]} sensitive', 28)
        d.text(xmax, label_y, "10 resistant", 28, anchor="end")
        endpoint = rx + resistant_width / 2
        leader_y = label_y - 30
        d.path([(endpoint, y + h / 2), (endpoint, leader_y - 4), (472, leader_y - 4), (472, leader_y)], p["resistantEdge"], 1.8, element_id=f"{name}-resistant-leader")
    d.rect(24, 703, 552, 497, p["panel"], p["border"], 16)
    d.text(48, coords["fractionPlot"]["headerY"], "Resistant fraction", 30, weight="bold")
    d.text(48, coords["fractionPlot"]["explicitLimitY"], "Percentage axis: 0–10%", 28)
    fraction = coords["fractionPlot"]
    for tick in data["plotAxes"]["resistantFraction"]["ticks"]:
        x = Fraction(xmin) + Fraction(xmax - xmin) * tick / 10
        d.path([(x, 864), (x, fraction["axisY"])], p["grid"], 1)
        d.path([(x, fraction["axisY"]), (x, fraction["axisY"] + 7)])
        d.text(x, fraction["axisY"] + 36, f"{tick}%", 26, anchor="middle")
    d.path([(xmin, fraction["axisY"]), (xmax, fraction["axisY"])])
    for name, state, formula in zip(("before", "after"), (data["before"], data["after"]), data["fractions"]):
        y = fraction["rowCenters"][name]
        h = fraction["barHeight"]
        fy = fraction["beforeFormulaY" if name == "before" else "afterFormulaY"]
        d.text(48, fy, name.capitalize(), 28, weight="bold")
        d.text(xmax, fy, formula, 28, anchor="end")
        percentage = Fraction(state["resistant"], state["total"]) * 100
        width = Fraction(xmax - xmin) * percentage / 10
        d.rect(xmin, y - h / 2, width, h, p["resistant"], p["resistantEdge"], element_id=f"{name}-resistant-fraction")
        d.rect(xmin, y - h / 2, width, h, "url(#resistant-hatch)")
    d.rect(24, 1210, 552, 67, p["takeaway"], radius=12)
    d.text(300, coords["ending"]["takeawayY"], "Resistant count stays 10", 30, anchor="middle", weight="bold")
    d.text(40, coords["ending"]["assumptionsY"][0], "No division or state change", 28)
    d.text(40, coords["ending"]["assumptionsY"][1], "No new mutation is established", 28)
    d.text(40, coords["ending"]["assumptionsY"][2], "Exact counts; not clinical results", 26, quiet=True)
    return d.finish()


def main() -> None:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--input-dir", type=Path, default=Path(__file__).resolve().parent)
    parser.add_argument("--output-dir", type=Path, required=True)
    args = parser.parse_args()
    data_path = args.input_dir / "chart-data-draft.json"
    coords_path = args.input_dir / "coordinate-specification-draft.json"
    data_bytes, coords_bytes = data_path.read_bytes(), coords_path.read_bytes()
    assert sha256(data_bytes).hexdigest() == "342bf930a3f174a49e9fc3180a34959514db6c72b85ff38568a6d084f77e7eb8"
    assert sha256(coords_bytes).hexdigest() == "2261298e3625b9c0c0ccb6cb8947e78000f1b59c49e71a1292338bc0b4ea56e3"
    data, coords = json.loads(data_bytes), json.loads(coords_bytes)
    checks = validate(data, coords)
    checks["inputHashes"] = {"chart-data-draft.json": sha256(data_bytes).hexdigest(), "coordinate-specification-draft.json": sha256(coords_bytes).hexdigest()}
    output = args.output_dir
    output.mkdir(parents=True, exist_ok=False)
    for theme in ("light", "dark"):
        (output / f"{STEM}-{theme}.svg").write_bytes(render(theme, data, coords, checks))
    (output / "chart-checks.json").write_text(json.dumps(checks, indent=2, ensure_ascii=False) + "\n", encoding="utf-8")
    (output / "chart-data.json").write_bytes(data_bytes)
    (output / "coordinate-specification.json").write_bytes(coords_bytes)


if __name__ == "__main__":
    main()
