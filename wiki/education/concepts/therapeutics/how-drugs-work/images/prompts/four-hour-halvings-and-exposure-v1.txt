def number(x):
    return f'{x:.6f}'.rstrip('0').rstrip('.')

def xy(t, c):
    return (PLOT['left']+(PLOT['right']-PLOT['left'])*t/12,
            PLOT['bottom']-(PLOT['bottom']-PLOT['top'])*c/100)

def validate(data):

    assert data['halfLife']['value']==4 and data['halfLife']['units']=='hours'
    assert len(data['continuousCurvePoints'])==301
    for i, point in enumerate(data['continuousCurvePoints']):
        assert math.isclose(point['timeHours'], i/25, abs_tol=1e-12)
        assert math.isclose(point['concentrationNgPerMl'], 100*2**(-point['timeHours']/4), abs_tol=1e-12)
    assert [(p['timeHours'],p['concentrationNgPerMl']) for p in data['anchors']]==[(0,100),(4,50),(8,25),(12,12.5)]
