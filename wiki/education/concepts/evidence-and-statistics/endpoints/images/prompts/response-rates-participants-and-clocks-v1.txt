===== EXACT ORIGINAL VECTOR PROGRAM =====

#!/usr/bin/env python3
"""Original editable vector authorship; standard Python only, no model/images."""
from pathlib import Path
from html import escape
import json
import sys

ROOT = Path(__file__).resolve().parent
DATA = json.loads((ROOT/'chart-data.json').read_text())
SPEC = json.loads((ROOT/'chart-spec.json').read_text())
STEM = 'response-rates-participants-and-clocks-v1'
PALETTES = {
    'light': {'ground':'#FBF8F1','panel':'#FFFFFF','ink':'#21333E','muted':'#425A68','edge':'#CBD3D4','groups':['#2C657E','#705E8C','#8B6541','#596D82','#5C6569'],'check':'#FFFFFF'},
    'dark': {'ground':'#17252D','panel':'#21323D','ink':'#F0F2EF','muted':'#CDD8DD','edge':'#526976','groups':['#91CEE0','#C3B3D9','#D8B58B','#B7CCE1','#B9C7CE'],'check':'#17252D'}
}

def render(theme):
    p=PALETTES[theme]
    elements=[]
    def text(x,y,content,size=32,weight=400,color=None):
        elements.append(f'<text x="{x}" y="{y}" font-size="{size}" font-weight="{weight}" fill="{color or p["ink"]}">{escape(content)}</text>')
    def shape(cx,cy,group,counted,legend=False):
        fill = p['groups'][group] if counted else p['panel']
        stroke = p['groups'][group]
        common=f'fill="{fill}" stroke="{stroke}" stroke-width="3"'
        # Category geometry remains recognizable without color.
        if group==0:
            elem=f'<rect x="{cx-21}" y="{cy-21}" width="42" height="42" rx="8" {common}/>'
        elif group==1:
            elem=f'<rect x="{cx-24}" y="{cy-16}" width="48" height="32" rx="16" {common}/>'
        elif group==2:
            elem=f'<path d="M {cx} {cy-24} L {cx+24} {cy} L {cx} {cy+24} L {cx-24} {cy} Z" {common}/>'
        elif group==3:
            elem=f'<path d="M {cx-23} {cy} L {cx-12} {cy-21} L {cx+12} {cy-21} L {cx+23} {cy} L {cx+12} {cy+21} L {cx-12} {cy+21} Z" {common}/>'
        else:
            elem=f'<circle cx="{cx}" cy="{cy}" r="21" {common}/>'
        if legend:
            elements.append(f'<g transform="translate({cx} {cy}) scale(0.72) translate({-cx} {-cy})">')
        elements.append(elem)
        if legend:
            elements.append('</g>')
        if not legend:
            ink=p['check'] if counted else stroke
            if counted:
                elements.append(f'<path d="M {cx-10} {cy} l 7 7 l 14 -16" fill="none" stroke="{ink}" stroke-width="4" stroke-linecap="round" stroke-linejoin="round"/>')
            else:
                elements.append(f'<path d="M {cx-10} {cy} h 20" fill="none" stroke="{ink}" stroke-width="4" stroke-linecap="round"/>')
    elements.append(f'<rect width="780" height="1500" rx="20" fill="{p["ground"]}"/>')
    text(36,56,'Same 20, different rules',44,700)
    text(36,104,'Fictional study • all 20 assessed',32,color=p['muted'])
    text(36,145,'0 unassessed • each mark = 1 person',32,color=p['muted'])
    group_order=[]
    for i,g in enumerate(DATA['groups']): group_order.extend([i]*g['count'])
    positions=[(76+(i%10)*70, i//10) for i in range(20)]
    for panel,(endpoint,label,y,h) in enumerate(zip(DATA['endpoints'],SPEC['labels'],[178,482,786],[284,284,340])):
        elements.append(f'<g id="{endpoint["key"]}" data-denominator="20" data-numerator="{endpoint["count"]}">')
        elements.append(f'<rect x="24" y="{y}" width="732" height="{h}" rx="22" fill="{p["panel"]}" stroke="{p["edge"]}" stroke-width="2"/>')
        text(44,y+44,label['heading'],36,650)
        text(44,y+88,label['result'],40,700)
        if panel<2:
            text(44,y+129,label['rule'],32)
            rowbase=y+162
        else:
            text(44,y+129,'Confirmed CR + PR OR',32)
            text(44,y+165,'SD ≥ 24 weeks',32)
            rowbase=y+198
        for i,((x,row),g) in enumerate(zip(positions,group_order)):
            counted=DATA['groups'][g]['qualifies'][endpoint['key']]
            elements.append(f'<g class="participant" data-slot="{i+1}" data-group="{DATA["groups"][g]["key"]}" data-counted="{str(counted).lower()}">')
            shape(x,rowbase+row*52,g,counted)
            elements.append('</g>')
        clock=['Follow-up window not supplied','Fixed assessment: week 8','24-week threshold on SD only'][panel]
        text(44,y+h-19,clock,30,color=p['muted'])
        elements.append('</g>')
    text(36,1170,'The same groups in every panel',32,650)
    legend=['2 confirmed complete responses (CR)','4 confirmed partial responses (PR)','3 week-8 SD; qualifies through week 24','5 other week-8 stable disease (SD)','6 other assessed outcomes']
    for i,txt in enumerate(legend):
        cy=1208+i*39
        elements.append('<g class="legend-item">')
        shape(58,cy-10,i,True,legend=True)
        text(98,cy,txt,30)
        elements.append('</g>')
    text(36,1430,'✓ Counts here • — does not count here',30)
    text(36,1474,'All 20 stay in each denominator',32,650)
    description='Three independent endpoint charts repeat all twenty fictional assessed participants. ORR6/20=30%; week8DCR14/20=70%; protocol-definedCBR9/20=45%. The24week threshold applies only to stable disease; no individual history is implied.'
    return f'''<svg xmlns="http://www.w3.org/2000/svg" width="780" height="1500" viewBox="0 0 780 1500" role="img" aria-labelledby="title desc" data-theme="{theme}">
<title id="title">Same 20, different response rules</title>
<desc id="desc">{escape(description)}</desc>
<g font-family="Arial, Helvetica, sans-serif">
{chr(10).join(elements)}
</g>
</svg>
'''

if __name__=='__main__':
    out=Path(sys.argv[1]) if len(sys.argv)>1 else ROOT/'assets'
    out.mkdir(parents=True,exist_ok=True)
    for theme in PALETTES:
        (out/f'{STEM}-{theme}.svg').write_text(render(theme))
    print('Rendered original editable light/dark SVGs from shared geometry/data; palette-only variants.')
