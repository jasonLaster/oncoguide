#!/usr/bin/env python3
"""Original stdlib editable CPM/TPM chart. Execute only after peer/root approval."""
import argparse,hashlib,json
from fractions import Fraction
from html import escape
from pathlib import Path

STEM='equal-counts-different-denominators-v1'
PALETTES={
 'light':{'ground':'#fbf7ef','panel':'#fffdf8','ink':'#26343b','muted':'#45545c','rule':'#6f7b80','strip':'#e9eee7','a':'#9ebda5','b':'#9fb9cf','hatch':'#395b75','badgeInk':'#20343b'},
 'dark':{'ground':'#18232d','panel':'#22323d','ink':'#f1eee5','muted':'#d1d9db','rule':'#a3b1b7','strip':'#2f4148','a':'#95b59f','b':'#91b0ca','hatch':'#263f55','badgeInk':'#20343b'}}
def sha(b):return hashlib.sha256(b).hexdigest()
def frac(r):return Fraction(r['numerator'],r['denominator'])
def number(v):
    v=Fraction(v)
    if v.denominator==1:return str(v.numerator)
    # All quantitative coordinates in this exact toy are integers.
    raise ValueError('Unexpected noninteger coordinate; extend exact serialization explicitly')
def write_json(p,o):p.write_text(json.dumps(o,ensure_ascii=False,indent=2)+'\n')

def validate(data,c):
    assert data['id']=='next-gap-014' and data['fictional'] is True
    ts=data['transcripts'];assert [(x['id'],x['assignedFragments'],x['lengthKb']) for x in ts]==[('A',100,1),('B',100,2)]
    total=sum(x['assignedFragments'] for x in ts);rates=[Fraction(x['assignedFragments'],x['lengthKb']) for x in ts]
    assert total==data['countTotal']==200 and sum(rates)==data['rateTotalFragmentsPerKb']==150
    cpms=[Fraction(x['assignedFragments'],total)*1000000 for x in ts];tpms=[r/sum(rates)*1000000 for r in rates]
    assert cpms==[500000,500000] and tpms==[Fraction(2000000,3),Fraction(1000000,3)]
    ledger=[]
    for name,values in [('cpmPlot',cpms),('tpmPlot',tpms)]:
        plot=c[name];axis=data['plotAxes'][name[:3]]
        assert axis['minimum']==0 and axis['maximum']==1000000 and axis['ticks']==[0,500000,1000000]
        assert plot['xMin']==144 and plot['xMax']==552
        span=plot['xMax']-plot['xMin']
        for row,value in zip(plot['rows'],values):
            assert frac(row['value'])==value
            width=Fraction(span)*value/axis['maximum'];end=plot['xMin']+width
            assert frac(row['widthExact'])==width and frac(row['endXExact'])==end
            ledger.append({'panel':name,'id':row['id'],'value':{'numerator':value.numerator,'denominator':value.denominator},'display':row['display'],'barX':plot['xMin'],'barY':row['barY'],'barHeight':plot['barHeight'],'width':number(width),'endX':number(end)})
    assert c['canvas']=={'width':600,'height':1560,'svgViewBox':'0 0 600 1560'} and c['font']['minimumSize']>=26
    assert [(r['id'],r['values']) for r in c['inputLedger']['dataRows']]==[('A',[100,1,100]),('B',[100,2,50])]
    return ledger

def render(data,c,theme):
    p=PALETTES[theme];out=[]
    def emit(s):out.append(s)
    def text(x,y,s,size=28,anchor='start',weight=400,role='ink'):
        emit(f'<text x="{number(x)}" y="{number(y)}" font-size="{size}" text-anchor="{anchor}" font-weight="{weight}" fill="{p[role]}">{escape(str(s))}</text>')
    def rect(x,y,w,h,fill,stroke=None,radius=0):
        extra=f' stroke="{stroke}" stroke-width="1.5"' if stroke else ''
        emit(f'<rect x="{number(x)}" y="{number(y)}" width="{number(w)}" height="{number(h)}" rx="{radius}" fill="{fill}"{extra}/>')
    def line(x1,y1,x2,y2,width=1.5):emit(f'<line x1="{number(x1)}" y1="{number(y1)}" x2="{number(x2)}" y2="{number(y2)}" stroke="{p["rule"]}" stroke-width="{width}"/>')
    emit('<?xml version="1.0" encoding="UTF-8"?>')
    emit('<svg xmlns="http://www.w3.org/2000/svg" width="600" height="1560" viewBox="0 0 600 1560" role="img" aria-labelledby="title desc">')
    emit('<title id="title">Equal counts, different CPM and TPM denominators</title>')
    emit('<desc id="desc">One fictional two-transcript full-length library. A and B each have 100 assigned fragments, with lengths 1 and 2 kilobases and rates 100 and 50 fragments per kilobase. Simple CPM is 500,000 each. Toy TPM uses exact two-thirds and one-third, displayed approximately 666,667 and 333,333. Separate unit labels and denominators; no molecule, cell, protein or cross-sample inference.</desc>')
    emit('<defs>')
    emit('<pattern id="b-hatch" patternUnits="userSpaceOnUse" width="10" height="10">')
    emit(f'<rect width="10" height="10" fill="{p["b"]}"/>')
    emit(f'<path d="M-2,2 L2,-2 M0,10 L10,0 M8,12 L12,8" fill="none" stroke="{p["hatch"]}" stroke-width="1.2"/>')
    emit('</pattern></defs>')
    rect(0,0,600,1560,p['ground'])
    emit(f'<g font-family="{escape(c["font"]["family"],quote=True)}">')
    text(*c['header']['titlePosition'],c['header']['title'][0],size=32,anchor='middle',weight=700)
    for s,x,y in c['header']['assumptionLines']:text(x,y,s,size=26,anchor='middle',role='muted')
    tab=c['inputLedger'];rect(*tab['panel'],p['panel'],p['rule'],12);s,x,y=tab['title'];text(x,y,s,size=30,weight=700)
    s,x,y=tab['rule'];text(x,y,s,size=26,role='muted')
    columns=tab['columnCenters'];hr=tab['headerRows']
    for name,labels in tab['headers'].items():
        for index,s in enumerate(labels):text(columns[name],[hr['namesY'],hr['unitY'],hr['rateExtraUnitY']][index],s,size=26,anchor='middle',weight=700 if index==0 else 400,role='ink' if index==0 else 'muted')
    line(48,326,552,326)
    for r in tab['dataRows']:
        x=columns['transcript'];y=r['baselineY'];rect(x-25,y-24,50,33,p['a'] if r['id']=='A' else 'url(#b-hatch)',p['rule'],4);text(x,y,r['id'],size=28,anchor='middle',weight=700,role='badgeInk')
        for name,value in zip(['count','length','rate'],r['values']):text(columns[name],y,value,size=30,anchor='middle',weight=700)
    for strip in tab['totalStrips']:rect(*strip['rect'],p['strip'],radius=5);text(*strip['position'],strip['text'],size=26,weight=700)
    for key in ['cpmPlot','tpmPlot']:
        plot=c[key];rect(*plot['panel'],p['panel'],p['rule'],12);s,x,y=plot['title'];text(x,y,s,size=30,weight=700)
        s,x,y=plot['unitLabel'];text(x,y,s,size=26,role='muted')
        for r in plot['rows']:
            height=plot['barHeight'];x0=plot['xMin'];w=frac(r['widthExact']);y=r['barY']
            text(70,y+height-3,r['id'],size=28,anchor='middle',weight=700)
            text(x0,r['valueY'],r['display'],size=30,weight=700)
            rect(x0,y,w,height,p['a'] if r['id']=='A' else 'url(#b-hatch)',p['rule'],0)
        line(plot['xMin'],plot['axisY'],plot['xMax'],plot['axisY'],2)
        for x,value,display,anchor in zip(c['ticks']['positions'],c['ticks']['values'],c['ticks']['display'],c['ticks']['anchors']):
            assert x==plot['xMin']+(plot['xMax']-plot['xMin'])*Fraction(value,1000000)
            line(x,plot['axisY'],x,plot['axisY']+7,2);text(x,plot['tickY'],display,size=26,anchor=anchor,role='muted')
        strip=plot['denominatorStrip'];rect(*strip['rect'],p['strip'],radius=5)
        for s,x,y in strip['lines']:text(x,y,s,size=26,weight=700)
    for s,x,y in c['ending']['lines']:text(x,y,s,size=26,anchor='middle',role='muted')
    emit('</g></svg>')
    return ('\n'.join(out)+'\n').encode()

    a=argparse.ArgumentParser(description=__doc__);a.add_argument('--data',type=Path,required=True);a.add_argument('--coordinates',type=Path,required=True);a.add_argument('--output',type=Path,required=True);args=a.parse_args()
